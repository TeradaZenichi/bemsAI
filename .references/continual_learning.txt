\section{Estratégias de Continual Learning}

A seguir, apresentamos um panorama das principais estratégias de \textit{continual learning}, suas variações e referências clássicas para cada método.

\begin{enumerate}[label=\textbf{\arabic*.}]
    \item \textbf{Regularização de Parâmetros}
    \begin{itemize}
        \item \textbf{Elastic Weight Consolidation (EWC):} Penaliza alterações em pesos importantes para tarefas anteriores, protegendo o conhecimento adquirido~\cite{Kirkpatrick2017}.
        \item \textbf{Synaptic Intelligence (SI), Memory Aware Synapses (MAS):} Variam o cálculo de importância dos pesos~\cite{Zenke2017,Aljundi2018}.
        \item \textbf{Learning without Forgetting (LwF):} Usa distilação para forçar a rede a preservar o conhecimento das tarefas antigas~\cite{Li2017}.
    \end{itemize}

    \item \textbf{Replay (Relembrar Exemplos Antigos)}
    \begin{itemize}
        \item \textbf{Experience Replay:} Amostras passadas são armazenadas e misturadas ao treino atual~\cite{Rolnick2019}.
        \item \textbf{Generative Replay:} Um gerador (ex: GAN) simula experiências antigas~\cite{Shin2017}.
    \end{itemize}

    \item \textbf{Interleaving/Intercalamento}
    \begin{itemize}
        \item Amostras antigas e novas são intercaladas em cada lote de treino para minimizar o esquecimento~\cite{Rebuffi2017,Hayes2020}.
    \end{itemize}

    \item \textbf{Métodos Arquiteturais/Dinâmicos (Expansão do Modelo)}
    \begin{itemize}
        \item \textbf{Progressive Neural Networks:} Novos módulos são adicionados para cada tarefa~\cite{Rusu2016}.
        \item \textbf{Dynamically Expandable Networks (DEN):} Expansão dinâmica da arquitetura~\cite{Yoon2018}.
        \item \textbf{PackNet:} Pruning iterativo para liberar espaço para novas tarefas~\cite{Mallya2018}.
    \end{itemize}

    \item \textbf{Métodos Baseados em Máscara/Ativação}
    \begin{itemize}
        \item \textbf{Hard Attention to the Task (HAT):} Máscaras binárias ativam partes específicas da rede~\cite{Serra2018}.
        \item \textbf{Supermasks/Task-specific Masks:} Segmentação da rede por tarefa~\cite{Wortsman2020}.
    \end{itemize}

    \item \textbf{Métodos de Distilação}
    \begin{itemize}
        \item \textbf{Knowledge Distillation:} Força o modelo a manter saídas semelhantes ao anterior~\cite{Hinton2015}.
        \item \textbf{Learning without Forgetting (LwF):} Exemplo prático de distilação em continual learning~\cite{Li2017}.
    \end{itemize}

    \item \textbf{Meta-Learning}
    \begin{itemize}
        \item Técnicas como \textbf{Reptile} e \textbf{MAML} treinam modelos para rápida adaptação sem esquecimento~\cite{Finn2017,Nichol2018}.
    \end{itemize}

    \item \textbf{Data Augmentation Temporal}
    \begin{itemize}
        \item Mistura e alteração de experiências antigas para simular maior diversidade~\cite{Isele2018}.
    \end{itemize}

    \item \textbf{Métodos Específicos para RL}
    \begin{itemize}
        \item \textbf{Policy Consolidation} e \textbf{Policy Distillation} evitam esquecimento de políticas em RL~\cite{Schwarz2018,Rusu2016b}.
    \end{itemize}
\end{enumerate}
